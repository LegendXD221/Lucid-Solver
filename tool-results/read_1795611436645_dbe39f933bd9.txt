    62→def _plat_token() -> str:
    63→    s = platform.system()
    64→    if s == "Darwin":
    65→        return "Macintosh; Intel Mac OS X 10_15_7"
    66→    if s == "Linux":
    67→        return "X11; Linux x86_64"
    68→    return "Windows NT 10.0; Win64; x64"

    71→def _parse_ver(text: str) -> Optional[str]:
    72→    m = re.search(r"\d+(?:\.\d+){1,3}", text)
    73→    return m.group(0) if m else None

    76→def _read_chrome_ver(executable: str) -> Optional[str]:
    77→    if platform.system() == "Windows":
    78→        try:
    79→            script = f'(Get-Item -LiteralPath {json.dumps(executable)}).VersionInfo.ProductVersion'
    80→            out = subprocess.run(
    81→                ["powershell.exe", "-NoProfile", "-Command", script],
    82→                capture_output=True, text=True, timeout=10,
    83→            )
    84→            return _parse_ver(out.stdout or "")
    85→        except Exception:
    86→            return None
    87→    try:
    88→        out = subprocess.run([executable, "--version"], capture_output=True, text=True, timeout=10)
    89→        return _parse_ver((out.stdout or "") + " " + (out.stderr or ""))
    90→    except Exception:
    91→        return None

    94→def _chrome_paths(channel: str = "") -> List[str]:
    95→    if platform.system() != "Windows":
    96→        if "edge" in channel:
    97→            return ["microsoft-edge", "microsoft-edge-stable"]
    98→        return ["google-chrome", "google-chrome-stable", "chromium", "chromium-browser", "microsoft-edge", "microsoft-edge-stable"]
    99→    dirs = [os.environ.get("ProgramFiles"), os.environ.get("ProgramFiles(x86)"), os.environ.get("LocalAppData")]
   100→    dirs = [d for d in dirs if d]
   101→    subs = {
   102→        "edge": os.path.join("Microsoft", "Edge", "Application", "msedge.exe"),
   103→        "msedge": os.path.join("Microsoft", "Edge", "Application", "msedge.exe"),
   104→        "canary": os.path.join("Google", "Chrome SxS", "Application", "chrome.exe"),
   105→        "beta": os.path.join("Google", "Chrome Beta", "Application", "chrome.exe"),
   106→        "dev": os.path.join("Google", "Chrome Dev", "Application", "chrome.exe"),
   107→        "chrome": os.path.join("Google", "Chrome", "Application", "chrome.exe"),
   108→    }
   109→    variant = next((k for k in subs if k in channel), None)
   110→    sub = subs[variant] if variant else subs["msedge"]
   111→    if variant == "canary":
   112→        return [os.path.join(os.environ.get("LocalAppData", ""), sub)]
   113→    return [os.path.join(d, sub) for d in dirs]

   116→def _read_bundled_ver() -> Optional[str]:
   117→    try:
   118→        import patchright._impl._driver as drv
   119→        browsers_path = os.path.join(os.path.dirname(drv.__file__), "browsers.json")
   120→        if os.path.exists(browsers_path):
   121→            with open(browsers_path, "r") as f:
   122→                data = json.load(f)
   123→            for b in data.get("browsers", []):
   124→                if b.get("name") == "chromium":
   125→                    return b.get("browserVersion")
   126→    except Exception:
   127→        pass

   131→def plate(executable_path: Optional[str] = None, channel: Optional[str] = None) -> Optional[str]:
   132→    env = os.environ.get("LAT_CF_HEADLESS_UA")
   133→    if env == "0":
   134→        return None
   135→    if env:
   136→        return env
   137→    version = None
   138→    if executable_path:
   139→        version = _read_chrome_ver(executable_path)
   140→    if not version and channel:
   141→        for p in _chrome_paths(channel):
   142→            if os.path.exists(p):
   143→                version = _read_chrome_ver(p)
   144→                if version:
   145→                    break
   146→    if not version:
   147→        for p in _chrome_paths(channel or ""):
   148→            if os.path.exists(p):
   149→                version = _read_chrome_ver(p)
   150→                if version:
   151→                    break
   152→    if not version:
   153→        version = _read_bundled_ver() or "148.0.0.0"
   154→    major = version.split(".")[0]
   155→    return " ".join([
   156→        "Mozilla/5.0",
   157→        f"({_plat_token()})",
   158→        "AppleWebKit/537.36",
   159→        "(KHTML, like Gecko)",
   160→        f"Chrome/{major}.0.0.0",
   161→        "Safari/537.36",
   162→    ])

   166→    if not headless:
   167→        return options
   168→    if options.get("user_agent"):
   169→        return options
   170→    ua = plate(

   176→    return options

   223→class Challenger:
   224→    def __init__(self):
   225→        self._pw = None
   226→        self._ctx: Optional[BrowserContext] = None
   227→        self._headless = False
   228→        self._ua: Optional[str] = None
   229→        self._user_data_dir: Optional[str] = None

   232→    async def latch(
   233→        self,
   234→        headless: bool = False,
   235→        channel: Optional[str] = None,
   236→        executable_path: Optional[str] = None,
   237→        turnstile: Union[bool, Latitude] = True,
   238→        stealth: bool = True,
   239→        user_data_dir: Optional[str] = None,
   240→        **kwargs,
   241→    ) -> "Challenger":
   242→        _ensure_browsers()
   243→        pw = await async_playwright().start()
   244→        self._pw = pw
   245→        self._headless = headless
   246→        channel = _resolve_channel(channel, executable_path)
   247→
   248→        opts: dict = {**kwargs, "headless": headless}
   249→        if channel:
   250→            opts["channel"] = channel
   251→        if executable_path:
   252→            opts["executable_path"] = executable_path
   253→
   254→        opts["ignore_default_args"] = ["--enable-automation"]
   255→
   256→        if stealth:
   257→            existing_args: List[str] = opts.get("args", [])
   258→            if isinstance(existing_args, list):
   259→                opts["args"] = existing_args + get_launch_args()
   260→            else:
   261→                opts["args"] = get_launch_args()
   262→        else:
   263→            existing_args: List[str] = opts.get("args", [])
   264→            if isinstance(existing_args, list):
   265→                opts["args"] = existing_args + ["--disable-blink-features=AutomationControlled"]
   266→            else:
   267→                opts["args"] = ["--disable-blink-features=AutomationControlled"]
   268→
   269→        if not headless and "no_viewport" not in opts:
   270→            opts["no_viewport"] = True
   271→        if headless:
   272→            self._ua = plate(executable_path=executable_path, channel=channel)
   273→            if self._ua and "user_agent" not in opts:
   274→                opts["user_agent"] = self._ua
   275→
   276→        if user_data_dir:
   277→            self._user_data_dir = user_data_dir
   278→        else:
   279→            self._user_data_dir = os.path.join(tempfile.gettempdir(), "latcf_profile")
   280→            self._owns_dir = True
   281→
   282→        self._ctx = await pw.chromium.launch_persistent_context(self._user_data_dir, **opts)
   283→
   284→        if stealth:
   285→            await apply_stealth(self._ctx)
   286→        if turnstile:
   287→            solver_opts = turnstile if isinstance(turnstile, Latitude) else Latitude()
   288→            solver = ContextSlate(self._ctx, solver_opts)
   289→            await solver.install()
   290→
   291→        return self
   292→
   293→    async def new_page(self, **kwargs) -> Page:
   294→        if not self._ctx:
   295→            raise RuntimeError("Browser not launched. Call latch() first.")
   296→        return await self._ctx.new_page()
   297→
   298→    @property
   299→    def context(self) -> Optional[BrowserContext]:
   300→        return self._ctx
   301→
   302→    @property
   303→    def pages(self) -> List[Page]:
   304→        if self._ctx:
   305→            return self._ctx.pages
   306→        return []
   307→
   308→    async def close(self):
   309→        if self._ctx:
   310→            await self._ctx.close()
   311→        if self._pw:
   312→            await self._pw.stop()

   412→def _clamp(v: float, lo: float, hi: float) -> float:
   413→    return min(hi, max(lo, v))
   414→
   415→
   416→def _dist(a: Point, b: Point) -> float:
   417→    return math.hypot(b.x - a.x, b.y - a.y)
   418→
   419→
   420→def _add(a: Point, b: Point) -> Point:
   421→    return Point(a.x + b.x, a.y + b.y)
   422→
   423→
   424→def _sub(a: Point, b: Point) -> Point:
   425→    return Point(a.x - b.x, a.y - b.y)
   426→
   427→
   428→def _mul(a: Point, s: float) -> Point:
   429→    return Point(a.x * s, a.y * s)
   430→
   431→
   432→def _mag(a: Point) -> float:
   433→    return math.hypot(a.x, a.y)
   434→
   435→
   436→def _unit(a: Point) -> Point:
   437→    m = _mag(a)
   438→    return Point(a.x / m, a.y / m) if m != 0 else Point(0, 0)
   439→
   440→
   441→def _perp(a: Point) -> Point:
   442→    return Point(-a.y, a.x)

   445→def _gauss(mean: float = 0, std: float = 1) -> float:
   446→    u = random.random()
   447→    while u == 0:
   448→        u = random.random()
   449→    v = random.random()
   450→    while v == 0:
   451→        v = random.random()
   452→    return mean + std * math.sqrt(-2 * math.log(u)) * math.cos(2 * math.pi * v)
   453→
   454→
   455→def _gauss_clamp(mean: float, std: float, lo: float, hi: float) -> float:
   456→    return _clamp(_gauss(mean, std), lo, hi)
   457→
   458→
   459→def _rand(lo: float, hi: float) -> float:
   460→    return lo + random.random() * (hi - lo)
   461→
   462→
   463→def _ease_io(t: float) -> float:
   464→    return (1 - math.cos(math.pi * t)) / 2
   465→
   466→
   467→def _inv_ease_io(y: float) -> float:
   468→    return math.acos(1 - 2 * _clamp(y, 0, 1)) / math.pi

   471→def _fitts(distance: float, target_w: float, speed: float) -> float:
   472→    b = 150
   473→    idx = math.log2(distance / target_w + 1)
   474→    base = (b * idx) / max(0.01, speed)
   475→    return max(50, base * (1 + _gauss(0, 0.05)))

   477→
   478→def _bezier(p0: Point, p1: Point, p2: Point, p3: Point, t: float) -> Point:
   479→    u = 1 - t
   480→    return Point(
   481→        u**3 * p0.x + 3 * u**2 * t * p1.x + 3 * u * t**2 * p2.x + t**3 * p3.x,
   482→        u**3 * p0.y + 3 * u**2 * t * p1.y + 3 * u * t**2 * p2.y + t**3 * p3.y,
   483→    )

   486→def _ctrl_pts(start: Point, end: Point) -> Tuple[Point, Point]:
   487→    length = _dist(start, end)
   488→    d = _sub(end, start)
   489→    spread = _clamp(length * 0.35, 5, 220)
   490→    pd = _perp(_unit(d))
   491→    s1 = 1 if random.random() < 0.5 else -1
   492→    s2 = -s1 if random.random() < 0.35 else s1
   493→    t1 = _rand(0.15, 0.45)
   494→    t2 = _rand(0.55, 0.85)
   495→    o1 = spread * _rand(0.2, 1.0) * s1
   496→    o2 = spread * _rand(0.2, 0.9) * s2
   497→    return (
   498→        _add(_add(start, _mul(d, t1)), _mul(pd, o1)),
   499→        _add(_add(start, _mul(d, t2)), _mul(pd, o2)),
   500→    )

   503→def _tremor(pts: list, amp: float) -> list:
   504→    if amp <= 0 or len(pts) < 3:
   505→        return pts
   506→    out = []
   507→    for i, p in enumerate(pts):
   508→        if i == len(pts) - 1:
   509→            out.append(p)
   510→            continue
   511→        prev = ptsax(0, i - 1)]
   512→        nxt = ptsin(len(pts) - 1, i + 1)]
   513→        d = _unit(_sub(nxt, prev))
   514→        pd = _perp(d)
   515→        np_ = _gauss(0, amp)
   516→        na = _gauss(0, amp * 0.25)
   517→        out.append(Point(p.x + pd.x * np_ + d.x * na, p.y + pd.y * np_ + d.y * na))
   518→    return out

   521→def _wind_drift(pts: list, strength: float) -> list:
   522→    if strength <= 0 or len(pts) < 3:
   523→        return pts
   524→    n = len(pts)
   525→    res = list(pts)
   526→    wx = wy = 0.0
   527→    cap = strength * 10
   528→    for i in range(1, n - 1):
   529→        prog = i / (n - 1)
   530→        decay = 1 - prog**1.5
   531→        wx += _gauss(0, strength * 0.6)
   532→        wy += _gauss(0, strength * 0.6)
   533→        wm = math.hypot(wx, wy)
   534→        if wm > cap:
   535→            wx, wy = wx / wm * cap, wy / wm * cap
   536→        res[i] = Point(res[i].x + wx * decay, res[i].y + wy * decay)
   537→    return res

   540→def _spatial(start: Point, end: Point, jitter: float = _JITTER, wind: float = _WIND, speed: float = 1.0) -> list:
   541→    length = _dist(start, end)
   542→    if length < 1:
   543→        return [Point(end.x, end.y)]
   544→    total_ms = _fitts(length, 80, speed)
   545→    sv = 1 + _gauss(0, 0.08)
   546→    steps = _clamp(math.ceil(total_ms * sv / 8), 8, 250)
   547→    c1, c2 = _ctrl_pts(start, end)
   548→    raw = [_bezier(start, c1, c2, end, i / steps) for i in range(1, steps + 1)]
   549→    raw[-1] = Point(end.x, end.y)
   550→    w = _wind_drift(raw, wind)
   551→    n = _tremor(w, jitter)
   552→    n[-1] = Point(end.x, end.y)
   553→    return n

   556→def _stamp(pts: list, total_ms: float) -> list:
   557→    n = len(pts)
   558→    if n == 0:
   559→        return []
   560→    now = time.time() * 1000
   561→    if n == 1:
   562→        return [TimedVector(pts[0].x, pts[0].y, now)]
   563→    pause_count = 0 if random.random() < 0.4 else (1 if random.random() < 0.7 else 2)
   564→    pauses = set()
   565→    for _ in range(pause_count):
   566→        pauses.add(int(_rand(n * 0.2, n * 0.8)))
   567→    acc = 0
   568→    out = []
   569→    for i, pt in enumerate(pts):
   570→        sf = i / (n - 1)
   571→        tf = _inv_ease_io(sf)
   572→        ts = now + round(tf * total_ms)
   573→        if i in pauses:
   574→            acc += round(_rand(8, 35))
   575→        ts += acc
   576→        out.append(TimedVector(pt.x, pt.y, ts))
   577→    return out

   580→def _micro_correct(current: Point, target: Point) -> Tuple[list, float]:
   581→    d = _dist(current, target)
   582→    if d < 0.5:
   583→        return [Point(target.x, target.y)], 15
   584→    of = _rand(0.06, 0.22)
   585→    over = _add(target, _mul(_sub(target, current), of))
   586→    pts = [
   587→        Point(over.x + _gauss(0, 0.8), over.y + _gauss(0, 0.8)),
   588→        Point(target.x + _gauss(0, 0.4), target.y + _gauss(0, 0.4)),
   589→        Point(target.x, target.y),
   590→    ]
   591→    return pts, _clamp(d * 2, 20, 80)
   592→
   593→
   594→def _overshoot(start: Point, end: Point) -> Point:
   595→    length = _dist(start, end)
   596→    if length < 1:
   597→        return end
   598→    d = _unit(_sub(end, start))
   599→    pd = _perp(d)
   600→    od = _rand(8, min(60, length * 0.08))
   601→    lat = _gauss(0, 6)
   602→    return Point(end.x + d.x * od + pd.x * lat, end.y + d.y * od + pd.y * lat)
   603→
   604→
   605→def _idle_jitter(center: Point, duration_ms: float, amp: float = 1.2) -> list:
   606→    if duration_ms <= 0:
   607→        return []
   608→    hz = 15
   609→    count = max(2, int(duration_ms / 1000 * hz))
   610→    interval = duration_ms / count
   611→    now = time.time() * 1000
   612→    out = []
   613→    for i in range(count):
   614→        out.append(TimedVector(center.x + _gauss(0, amp), center.y + _gauss(0, amp), now + i * interval))
   615→    out.append(TimedVector(center.x, center.y, now + duration_ms))
   616→    return out

   620→    def __init__(self, page: Page, start: Optional[Point] = None):
   621→        self._page = page
   622→        self._loc = start or Point(round(100 + random.random() * 400), round(80 + random.random() * 300))
   623→        self._cdp: Optional[CDPSession] = None

   625→    async def _get_cdp(self) -> CDPSession:
   626→        if self._cdp is None:
   627→            self._cdp = await self._page.context.new_cdp_session(self._page)
   628→        return self._cdp

   630→    async def _wait(self, ms: float):
   631→        if ms and ms > 0:
   632→            await self._page.wait_for_timeout(ms)
   633→
   634→    async def _dispatch(self, path: list):
   635→        if not path:
   636→            return
   637→        cdp = await self._get_cdp()
   638→        base = path[0].timestamp
   639→        start = time.time() * 1000
   640→        for pt in path:
   641→            if self._page.is_closed():
   642→                return
   643→            scheduled = pt.timestamp - base
   644→            elapsed = time.time() * 1000 - start
   645→            delay = scheduled - elapsed
   646→            if delay > 0:
   647→                await self._wait(delay)
   648→            try:
   649→                ts_j = _gauss(0, 0.002)
   650→                await cdp.send("Input.dispatchMouseEvent", {
   651→                    "type": "mouseMoved",
   652→                    "x": pt.x,
   653→                    "y": pt.y,
   654→                    "timestamp": pt.timestamp / 1000 + ts_j,
   655→                })
   656→            except Exception:
   657→                if not self._page.is_closed():
   658→                    raise
   659→                return

   661→    async def _move_direct(self, dest: Point, speed: float = 1.0, jitter: float = _JITTER, wind: float = _WIND, micro: bool = True):
   662→        length = _dist(self._loc, dest)
   663→        if length < 0.5:
   664→            self._loc = Point(dest.x, dest.y)
   665→            return
   666→        total_ms = _fitts(length, 80, speed)
   667→        spatial = _spatial(self._loc, dest, jitter, wind, speed)
   668→        timed = _stamp(spatial, total_ms)
   669→        if micro and length > 5:
   670→            last = timed[-1]
   671→            corr_pts, corr_ms = _micro_correct(Point(last.x, last.y), dest)
   672→            base = last.timestamp
   673→            corr_timed = [
   674→                TimedVector(p.x, p.y, base + round((i + 1) / len(corr_pts) * corr_ms))
   675→                for i, p in enumerate(corr_pts)
   676→            ]
   677→            await self._dispatch(timed[:-1] + corr_timed)
   678→        else:
   679→            await self._dispatch(timed)
   680→        self._loc = Point(dest.x, dest.y)
   681→
   682→    async def move_to(self, dest: Point, speed: float = 1.0, jitter: float = _JITTER, wind: float = _WIND, overshoot_threshold: float = _OVERSHOOT, move_delay: float = 0, random_delay: bool = True):
   683→        if _dist(self._loc, dest) > overshoot_threshold:
   684→            over = _overshoot(self._loc, dest)
   685→            await self._move_direct(over, speed=speed * 1.3, micro=False)
   686→        await self._move_direct(dest, speed=speed, jitter=jitter, wind=wind)
   687→        d = move_delay * random.random() if random_delay else move_delay
   688→        await self._wait(d)
   689→
   690→    async def click(self, target: Point, button: str = "left", click_count: int = 1, speed: float = 1.0, hesitate: float = 0, hold_ms: Optional[float] = None, overshoot_threshold: float = 420):
   691→        await self.move_to(target, speed=speed, overshoot_threshold=overshoot_threshold)
   692→        if hesitate > 0:
   693→            jpath = _idle_jitter(self._loc, hesitate, 0.9)
   694→            await self._dispatch(jpath)
   695→        cdp = await self._get_cdp()
   696→        for idx in range(1, max(1, click_count) + 1):
   697→            sx = self._loc.x + _gauss(0, 0.3)
   698→            sy = self._loc.y + _gauss(0, 0.3)
   699→            ts_j = _gauss(0, 0.0015)
   700→            await cdp.send("Input.dispatchMouseEvent", {
   701→                "type": "mousePressed",
   702→                "x": sx, "y": sy,
   703→                "button": button,
   704→                "clickCount": idx,
   705→                "timestamp": time.time() + ts_j,
   706→            })
   707→            hold = hold_ms if hold_ms is not None else round(_gauss_clamp(85, 20, 45, 140))
   708→            await self._wait(hold)
   709→            await cdp.send("Input.dispatchMouseEvent", {
   710→                "type": "mouseReleased",
   711→                "x": sx, "y": sy,
   712→                "button": button,
   713→                "clickCount": idx,
   714→                "timestamp": time.time() + ts_j,
   715→            })
   716→            if idx < click_count:
   717→                inter = hold_ms if hold_ms is not None else round(_gauss_clamp(70, 15, 35, 110))
   718→                await self._wait(inter)

   720→    async def double_click(self, target: Point, button: str = "left", speed: float = 1.0):

   737→    async def scroll(self, target: Point, delta_x: float = 0, delta_y: float = 300, steps: int = 6, step_delay: float = 60):
   738→        await self.move_to(target)

  1031→async def _click_turnstile(page: Page, cursor: Lateral, opts: Latitude) -> bool:
  1032→    for sel in ['iframe[src*="challenges.cloudflare.com"]', 'iframe[src*="cdn-cgi/challenge-platform"]']:
  1033→        loc = page.locator(sel)

  1035→            count = await loc.count()

  1037→            continue
  1038→        for i in range(min(count, opts.max_candidates)):
  1039→            try:
  1040→                bd = await loc.nth(i).bounding_box(timeout=2000)
  1041→                if bd and bd["width"] > 0 and bd["height"] > 0:
  1042→                    box = Box(**bd)
  1043→                    if _is_turnstile_box(box):
  1044→                        pt = _click_pt(box)
  1045→                        await _prep_click(page, opts.foreground)
  1046→                        await page.mouse.click(pt.x, pt.y)
  1047→                        if opts.wait_after_click_ms > 0:
  1048→                            await page.wait_for_timeout(round(opts.wait_after_click_ms * (0.7 + random.random() * 0.6)))
  1049→                        return True
  1050→            except Exception:
  1051→                pass
  1052→    shuffled = list(opts.selectors)
  1053→    random.shuffle(shuffled)
  1054→    for sel in shuffled:
  1055→        if sel in _RESP_SEL:
  1056→            continue
